-- Função para encontrar as frutas no mapa
local function GetFrutas()
    local list = {}
    -- Procuramos no workspace para encontrar as frutas que spawnaram
    for _, v in pairs(workspace:GetChildren()) do 
        if v:IsA("Tool") and v:FindFirstChild("Handle") and v.Name:lower():find("fruit") then 
            table.insert(list, v) 
        end 
    end
    return list
end

-- AUTO TWEEN CORRIGIDO E OTIMIZADO
local AutoOn = false
local TweenAtual = nil

local AutoBtn = Instance.new("TextButton", PopupFruta)

-- Função para calcular o tempo baseado em uma velocidade constante (ex: 50 studs por segundo)
local function getTweenTime(distance)
    return distance / 50 
end

AutoBtn.MouseButton1Click:Connect(function()
    AutoOn = not AutoOn
    AutoBtn.Text = AutoOn and "✈️ Auto Tween Fruta [ON]" or "✈️ Auto Tween Fruta [OFF]"
    AutoBtn.BackgroundColor3 = AutoOn and Color3.fromRGB(0,150,90) or Color3.fromRGB(35,20,65)
    
    if not AutoOn then
        if TweenAtual then TweenAtual:Cancel() end
        return
    end

    task.spawn(function()
        while AutoOn do
            local frutas = GetFrutas()
            local char = P.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")

            if #frutas > 0 and hrp then
                -- Encontra a fruta mais próxima
                table.sort(frutas, function(a, b)
                    local distA = (hrp.Position - a.Handle.Position).Magnitude
                    local distB = (hrp.Position - b.Handle.Position).Magnitude
                    return distA < distB
                end)

                local alvo = frutas[1]
                if alvo and alvo:FindFirstChild("Handle") then
                    local alvoPos = alvo.Handle.Position
                    local distancia = (hrp.Position - alvoPos).Magnitude

                    -- Se estiver longe, faz o Tween
                    if distancia > 5 then
                        if TweenAtual then TweenAtual:Cancel() end
                        
                        -- Criamos um Tween com tempo dinâmico para manter velocidade constante
                        local info = TweenInfo.new(getTweenTime(distancia), Enum.EasingStyle.Linear)
                        TweenAtual = T:Create(hrp, info, {CFrame = CFrame.new(alvoPos + Vector3.new(0, 3, 0))})
                        TweenAtual:Play()
                        
                        -- Espera o Tween chegar perto ou o tempo acabar
                        local tempoEsperado = getTweenTime(distancia)
                        local start = tick()
                        
                        -- Loop de espera inteligente para não travar o script
                        while tick() - start < tempoEsperado and AutoOn do
                            if (hrp.Position - alvoPos).Magnitude < 7 then break end
                            task.wait(0.1)
                        end
                    end

                    -- Simulação de Coleta (Touch)
                    if AutoOn and hrp then
                        pcall(function()
                            firetouchinterest(hrp, alvo.Handle, 0)
                            task.wait(0.1)
                            firetouchinterest(hrp, alvo.Handle, 1)
                        end)
                    end
                end
            end
            task.wait(0.5) -- Delay para não sobrecarregar o processador
        end
    end)
end)

local function OptimizeFPS()
    print("[DeepHat] Iniciando Otimização de FPS...")

    -- 1. Otimização de Iluminação (Reduz carga de sombras e reflexos)
    local Lighting = game:GetService("Lighting")
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e9 -- Aumenta o alcance para não processar neblina densa
    Lighting.Brightness = 1
    
    -- 2. Limpeza de Efeitos Visuais (Bloom, Blur, SunRays)
    for _, effect in pairs(Lighting:GetChildren()) do
        if effect:IsA("BloomEffect") or effect:IsA("BlurEffect") or effect:IsA("SunRaysEffect") or effect:IsA("ColorCorrectionEffect") then
            effect.Enabled = false
        end
    end

    -- 3. Otimização de Renderização de Partículas e Texturas
    -- Isso reduz o uso de VRAM e processamento de shaders
    task.spawn(function()
        for _, v in pairs(workspace:GetDescendants()) do
            -- Remove partículas que pesam no FPS
            if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") then
                v.Enabled = false
            end
            
            -- Remove texturas de alta resolução (ajuda em máquinas fracas)
            if v:IsA("Texture") or v:IsA("Decal") then
                v.Transparency = 0.5 -- Deixa semi-transparente para economizar render
            end
            
            -- Desativa reflexos em materiais metálicos
            if v:IsA("BasePart") then
                v.Material = Enum.Material.SmoothPlastic
                v.Reflectance = 0
            end
        end
    end)

    -- 4. Otimização de Som (Reduz uso de CPU em ambientes barulhentos)
    local SoundService = game:GetService("SoundService")
    for _, sound in pairs(game:GetDescendants()) do
        if sound:IsA("Sound") then
            sound.Volume = 0.5 -- Reduz volume global para economizar processamento de áudio
        end
    end

    -- 5. Configurações de Engine (Propriedades de renderização)
    settings().Rendering.QualityLevel = 1 -- Força nível de qualidade baixo
    
    print("[DeepHat] FPS Booster Ativado! Renderização otimizada.")
end

local function CreateFPSCounter()
    local FPSGui = Instance.new("ScreenGui")
    FPSGui.Name = "FPSMonitorGui"
    FPSGui.Parent = P:WaitForChild("PlayerGui")
    FPSGui.ResetOnSpawn = false

    local Container = Instance.new("Frame")
    Container.Size = UDim2.new(0, 120, 0, 40)
    Container.Position = UDim2.new(0, 10, 0, 10) -- Canto superior esquerdo
    Container.BackgroundColor3 = Color3.fromRGB(20, 10, 30)
    Container.BackgroundTransparency = 0.3
    Container.Parent = FPSGui
    
    local Corner = Instance.new("UICorner", Container)
    Corner.CornerRadius = UDim.new(0, 8)
    
    local Stroke = Instance.new("UIStroke", Container)
    Stroke.Thickness = 2
    Stroke.Color = Color3.fromRGB(120, 60, 220)
    Stroke.Transparency = 0.5

    local Label = Instance.new("TextLabel", Container)
    Label.Size = UDim2.new(1, 0, 1, 0)
    Label.BackgroundTransparency = 1
    Label.Text = "FPS: --"
    Label.TextColor3 = Color3.fromRGB(255, 255, 255)
    Label.Font = Enum.Font.GothamBold
    Label.TextSize = 14
    Label.TextXAlignment = Enum.TextXAlignment.Center
    Label.TextYAlignment = Enum.TextYAlignment.Center

    -- Lógica de Cálculo de FPS
    local fps = 0
    local frameCount = 0
    local lastUpdate = tick()

    RunService.RenderStepped:Connect(function(deltaTime)
        frameCount += 1
        
        -- Atualiza a cada 0.5 segundos para não ficar oscilando muito rápido
        if tick() - lastUpdate >= 0.5 then
            fps = math.floor(frameCount / (tick() - lastUpdate))
            lastUpdate = tick()
            frameCount = 0
            
            -- Muda a cor baseado no desempenho (Verde = Bom, Amarelo = Médio, Vermelho = Baixo)
            local color = Color3.fromRGB(0, 255, 150) -- Verde
            if fps < 40 then
                color = Color3.fromRGB(255, 80, 80) -- Vermelho
            elseif fps < 55 then
                color = Color3.fromRGB(255, 200, 50) -- Amarelo
            end
            
            Label.Text = "FPS: " .. tostring(fps)
            Label.TextColor3 = color
            
            -- Efeito visual de pulso ao atualizar
            local tween = T:Create(Label, TweenInfo.new(0.2), {TextSize = 16})
            tween:Play()
            tween.Completed:Wait()
            T:Create(Label, TweenInfo.new(0.2), {TextSize = 14}):Play()
        end
    end)

    return FPSGui
end

-- === LÓGICA DE DETECÇÃO ===

local function GetClosestPlayer()
    local Target = nil
    local MinDistance = Settings.FOV 

    for _, v in pairs(game.Players:GetPlayers()) do
        if v ~= Player and v.Character and v.Character:FindFirstChild(Settings.TargetPart) and v.Character:FindFirstChild("Humanoid") and v.Character.Humanoid.Health > 0 then
            
            -- Cálculo de distância entre você e o alvo
            local DistanceToPlayer = (Player.Character.HumanoidRootPart.Position - v.Character[Settings.TargetPart].Position).Magnitude
            
            -- Filtro de Distância (Agora com valor maior)
            if DistanceToPlayer <= Settings.MaxDistance then
                
                local ScreenPos, OnScreen = Camera:WorldToViewportPoint(v.Character[Settings.TargetPart].Position)
                
                if OnScreen then
                    local MousePos = Vector2.new(Mouse.X, Mouse.Y)
                    local MagnitudeOnScreen = (Vector2.new(ScreenPos.X, ScreenPos.Y) - MousePos).Magnitude
                    
                    if MagnitudeOnScreen < MinDistance then
                        Target = v
                        MinDistance = MagnitudeOnScreen
                    end
                end
            end
        end
    end
    return Target
end

-- === LOOP DE RENDERIZAÇÃO ===

RunService.RenderStepped:Connect(function()
    if Settings.AimbotEnabled then
        if not ClosestPlayer or not ClosestPlayer.Character or not ClosestPlayer.Character:FindFirstChild(Settings.TargetPart) or ClosestPlayer.Character.Humanoid.Health <= 0 then
            ClosestPlayer = GetClosestPlayer()
        else
            -- Verifica se o alvo fugiu da distância permitida
            local CurrentDist = (Player.Character.HumanoidRootPart.Position - ClosestPlayer.Character[Settings.TargetPart].Position).Magnitude
            if CurrentDist > Settings.MaxDistance then
                ClosestPlayer = nil
            end
        end

        if ClosestPlayer and ClosestPlayer.Character and ClosestPlayer.Character:FindFirstChild(Settings.TargetPart) then
            local TargetPos = ClosestPlayer.Character[Settings.TargetPart].Position
            local CurrentCFrame = Camera.CFrame
            local LookAtCFrame = CFrame.new(CurrentCFrame.Position, TargetPos)
            
            Camera.CFrame = CurrentCFrame:Lerp(LookAtCFrame, Settings.Smoothing)
        end
    end
end)
